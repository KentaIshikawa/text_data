真偽値とif文

真偽値は、if文の条件に使ったとき、trueだったら条件を通り、falseだったら通らない仕様になっている。

const check = true;

if(check){
    console.log('変数checkには、trueが代入されているので必ず条件は成り立つ');
}

const btn = document.querySelector('#btn');

btn.addEventListener('click', () => {
    if (check) {
        console.log('成り立つ');
        check = false;
    } else {
        console.log('成り立たない');
        check = true;
    }
});



const color_btn = document.querySelector('#color');
let color_check = true;
color_btn.addEventListener('click', () => {
    if (color_check) {
        
        const color_arr = ['青', '緑', '赤'];

        const color_class = ['blue', 'green', 'red'];

        const rand = Math.floor(Math.random() * color_arr.length);

        const text = document.querySelector('#text');
        text.classList = color_class[rand];

        text.innerHTML = '今日のあなたのラッキーカラーは、' + color_arr[rand] + 'です！';

        color_check = false;
    } else {
        alert('一回までです');
    }
});
